# Quijote amb turbines de punta en lloc de masses radials
import numpy as np
from scipy.integrate import odeint
import matplotlib.pyplot as plt

TWO_PI = 2*np.pi
RHO = 1.225
g = 9.81

class TipTurbine:
    """Turbina petita a la punta d'una aspa."""
    def __init__(self, a_t=1.0, Cp0=0.40, CT0=0.80, eta_t=0.85):
        self.a_t, self.Cp0, self.CT0, self.eta_t = a_t, Cp0, CT0, eta_t

    def set_beta(self, beta_deg):
        self.beta = np.deg2rad(beta_deg)
        self.Cp = self.Cp0 * np.cos(self.beta)
        self.CT = self.CT0 * np.cos(self.beta)

    def power(self, v_app):
        return self.eta_t * self.Cp * 0.5 * RHO * self.a_t * v_app**3

    def thrust(self, v_app):
        return self.CT * 0.5 * RHO * self.a_t * v_app**2

class QuijoteTipMill:
    def __init__(self, name, R, I_gen, omega_rated, beta_fric,
                 n_physical=1, delta_opt_deg=30.0):
        self.name, self.R, self.I_gen = name, R, I_gen
        self.omega_rated = omega_rated
        self.beta_fric = beta_fric
        self.n_physical = n_physical
        self.delta_opt = np.deg2rad(delta_opt_deg)
        self.N_blades = 3
        self.tips = [TipTurbine() for _ in range(self.N_blades)]

    def sector(self, theta):
        th = theta % TWO_PI
        if th < np.pi/3:      return 'consume',   0.0    # cara al vent
        elif th < np.pi:      return 'consume',   0.0
        elif th < 4*np.pi/3:  return 'compound',  90.0   # plomada + rumb
        else:                 return 'feather',   90.0

    def apply_window(self, theta_rot):
        """Aplica la finestra a cada turbina de punta."""
        for j, tip in enumerate(self.tips):
            th_j = (theta_rot + j*TWO_PI/3) % TWO_PI
            state, beta = self.sector(th_j)
            tip.set_beta(beta)
            tip.state = state

    def tip_torques(self, theta_rot, omega):
        """Retorna (tau_compounding, tau_drag, P_electrical)."""
        tau_comp = 0.0; tau_drag = 0.0; P_elec = 0.0
        for j, tip in enumerate(self.tips):
            th_j = theta_rot + j*TWO_PI/3
            # v_app: vent + gir
            v_app = np.sqrt(1.0**2 + (omega*self.R)**2) * 12.0
            F = tip.thrust(v_app) * self.n_physical
            P = tip.power(v_app) * self.n_physical
            # Component tangencial (compounding) si state == 'compound'
            if getattr(tip, 'state', 'feather') == 'compound':
                tau_comp += self.n_physical * self.R * F * np.cos(self.delta_opt)
            # Component radial (drag) sempre, però gran si 'consume'
            tau_drag -= self.n_physical * self.R * F * np.sin(self.delta_opt)
            P_elec += P
        return tau_comp, tau_drag, P_elec

    def gravity_torque(self, theta_rot):
        # Ja no hi ha masses; la gravetat actua sobre el rotor principal (petit)
        return 0.0

    def inertia(self):
        return self.I_gen

class NetworkTip:
    def __init__(self):
        self.C1 = QuijoteTipMill('C1', 40, 2e7, 1.5, 5e5, 1, delta_opt_deg=25)
        self.C2 = QuijoteTipMill('C2', 40, 2e7, 1.5, 5e5, 1, delta_opt_deg=25)
        self.A1 = QuijoteTipMill('A1', 20, 4e6, 2.0, 1.5e5, 2, delta_opt_deg=30)
        self.A2 = QuijoteTipMill('A2', 20, 4e6, 2.0, 1.5e5, 2, delta_opt_deg=30)
        self.A3 = QuijoteTipMill('A3', 20, 4e6, 2.0, 1.5e5, 2, delta_opt_deg=30)
        self.mills = [self.C1, self.C2, self.A1, self.A2, self.A3]
        self.A = np.array([[0,0,1,1,1],[0,0,1,1,1],
                           [1,1,0,1,1],[1,1,1,0,1],[1,1,1,1,0]], float)
        self.K = 0.5; self.K_bus = 1.0
        self.omega_elec = 5.0
        self.v_base = 12.0

    def rhs(self, s, t):
        n = 5
        theta = s[0:n]; omega = s[n:2*n]
        phi   = s[2*n:3*n]
        dtheta = omega.copy(); domega = np.zeros(n); dphi = np.zeros(n)
        phi_bus = np.angle(np.mean(np.exp(1j*phi[:2])))
        v = self.v_base * (1 + 0.1*np.sin(0.3*t))

        for i, mill in enumerate(self.mills):
            mill.apply_window(theta[i])
            tau_comp, tau_drag, P_elec = mill.tip_torques(theta[i], omega[i])
            I = mill.inertia()
            # Parell net: compounding - drag - fricció + Kuramoto
            t_k = sum(self.K*self.A[i, j]*np.sin(phi[j]-phi[i])
                      for j in range(n) if j != i)
            domega[i] = (tau_comp + tau_drag - mill.beta_fric*omega[i] + t_k) / I
            # Fase Kuramoto
            coupling = (self.K/n)*np.sum(self.A[i]*np.sin(phi - phi[i]))
            dphi[i] = self.omega_elec + coupling + self.K_bus*np.sin(phi_bus - phi[i])

        return np.concatenate([dtheta, domega, dphi])

def run(duration=60.0, dt=1e-3):
    net = NetworkTip(); n = 5
    theta0 = np.array([0.0, np.pi/3, np.pi/6, np.pi/2, 2*np.pi/3])
    omega0 = np.array([1.4, 1.6, 1.8, 2.0, 2.2])
    phi0 = np.array([0.0, 0.5, 1.0, 1.5, 2.0])
    s0 = np.concatenate([theta0, omega0, phi0])
    t = np.arange(0, duration, dt)
    sol = odeint(net.rhs, s0, t, rtol=1e-6, atol=1e-8)
    theta = sol[:, :n]; omega = sol[:, n:2*n]; phi = sol[:, 2*n:3*n]
    return net, t, theta, omega, phi

if __name__ == '__main__':
    net, t, theta, omega, phi = run(60.0)
    labels = ['C1','C2','A1','A2','A3']
    fig, ax = plt.subplots(2, 2, figsize=(14, 9))

    for i in range(5):
        ax[0,0].plot(t, omega[:,i], label=labels[i])
    ax[0,0].set_title('ω_i(t) — turbines de punta (compounding vs drag)')
    ax[0,0].legend(); ax[0,0].grid(alpha=0.3)

    r_ord = np.abs(np.mean(np.exp(1j*phi), axis=1))
    ax[0,1].plot(t, r_ord, color='purple')
    ax[0,1].axhline(0.9, ls='--', color='red')
    ax[0,1].set_title('Kuramoto r(t)'); ax[0,1].grid(alpha=0.3)

    # Estat de la turbina de punta C1 aspa 1
    states = []
    for th in theta[:,0]:
        st, _ = net.C1.sector(th)
        states.append({'consume': 1, 'compound': 2, 'feather': 0}[st])
    ax[1,0].plot(t, states, color='orange')
    ax[1,0].set_title('C1 aspa 1: estat turbina (0=feather, 1=consume, 2=compound)')
    ax[1,0].set_yticks([0,1,2]); ax[1,0].grid(alpha=0.3)

    # Potència elèctrica total de les turbines de punta
    P_tot = np.zeros(len(t))
    for k, th in enumerate(theta):
        for i, mill in enumerate(net.mills):
            mill.apply_window(th[i])
            _, _, P = mill.tip_torques(th[i], omega[k,i])
            P_tot[k] += P
    ax[1,1].plot(t, P_tot/1e3, color='green')
    ax[1,1].set_title('Potència elèctrica total de les turbines de punta (kW)')
    ax[1,1].set_xlabel('t (s)'); ax[1,1].grid(alpha=0.3)

    plt.tight_layout(); plt.savefig('tip_turbines.png', dpi=140); plt.show()

    print(f"r Kuramoto final: {r_ord[-1]:.3f}")
    print(f"ω final: {dict(zip(labels, np.round(omega[-1],3)))}")
    print(f"P_tips mitjana: {P_tot.mean()/1e3:.1f} kW")
