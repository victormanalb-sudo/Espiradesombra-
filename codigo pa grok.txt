# --- Parámetros geométricos ---
RADIO = 5.0  # Radio de giro (m)
ALTURA_INICIAL = 45.0  # Altura inicial del lastre (m)
ALTURA_FINAL = -45.0  # Altura final del lastre (m)
DELTA_H = ALTURA_INICIAL - ALTURA_FINAL  # Diferencia de altura (90 m)

# --- Parámetros de masa y densidad ---
M_LASTRE = 10.0  # Masa del lastre (kg)
RHO_LASTRE = 13600.0  # Densidad del lastre (mercurio, kg/m³)
RHO_FLUIDO = 1000.0  # Densidad del fluido (agua, kg/m³)
G = 9.81  # Aceleración gravitatoria (m/s²)

# --- Parámetros de energía ---
ETA_GEN = 0.85  # Eficiencia en generación
ETA_LIFT = 0.90  # Eficiencia en elevación
E_PERNO_UNITARIO = 1.5  # Energía por perno (J)

# --- Parámetros de simulación ---
DT = 0.01  # Paso de tiempo (s)

class Lastre:
    """
    Clase para modelar el lastre en la Máquina Kilómetro.
    
    Atributos:
        m (float): Masa del lastre (kg). Ej: 10 kg de mercurio.
        rho (float): Densidad del lastre (kg/m³). Ej: 13600 kg/m³ (mercurio).
        g (float): Aceleración gravitatoria (m/s²). Ej: 9.81 m/s².
        altura (float): Altura actual del lastre (m). Ej: +45 m o -45 m.
        energia_potencial (float): Energía potencial actual (J).
    
    Métodos:
        soltar(altura_final): Suelta el lastre y genera energía.
        subir(altura_inicial): Sube el lastre y consume energía.
    """
    
    def __init__(self, m, rho, g, altura_inicial):
        self.m = m
        self.rho = rho
        self.g = g
        self.altura = altura_inicial
        self.energia_potencial = self.m * self.g * self.altura
    
    def soltar(self, altura_final):
        """
        Suelta el lastre desde altura_inicial a altura_final.
        
        Args:
            altura_final (float): Altura final del lastre (m). Ej: -45 m.
        
        Returns:
            float: Energía generada al soltar el lastre (J).
                   Fórmula: E = m * g * Δh * η_gen
        
        Ejemplo:
            m = 10 kg, g = 9.81 m/s², Δh = 90 m, η_gen = 0.85
            E = 10 * 9.81 * 90 * 0.85 ≈ 7641 J
        """
        delta_h = self.altura - altura_final
        energia_generada = self.m * self.g * delta_h * ETA_GEN
        self.altura = altura_final
        self.energia_potencial = self.m * self.g * self.altura
        return energia_generada
    
    def subir(self, altura_inicial):
        """
        Sube el lastre desde altura_final a altura_inicial.
        
        Args:
            altura_inicial (float): Altura inicial del lastre (m). Ej: +45 m.
        
        Returns:
            float: Energía consumida al subir el lastre (J).
                   Fórmula: E = (m * g * Δh) / η_lift
        
        Ejemplo:
            m = 10 kg, g = 9.81 m/s², Δh = 90 m, η_lift = 0.90
            E = (10 * 9.81 * 90) / 0.90 ≈ 9810 J
        """
        delta_h = altura_inicial - self.altura
        energia_consumida = (self.m * self.g * delta_h) / ETA_LIFT
        self.altura = altura_inicial
        self.energia_potencial = self.m * self.g * self.altura
        return energia_consumida


class Objeto:
    """
    Clase para modelar el objeto en el DKP.
    
    Atributos:
        radio (float): Radio de giro (m). Ej: 5 m.
        altura (float): Altura actual del objeto (m). Ej: +45 m, 0 m.
        angulo (float): Ángulo actual del objeto (rad).
        energia_cinetica (float): Energía cinética actual (J).
    
    Métodos:
        girar(vueltas): Gira el objeto y genera energía cinética.
    """
    
    def __init__(self, radio, altura_inicial):
        self.radio = radio
        self.altura = altura_inicial
        self.angulo = 0.0
        self.energia_cinetica = 0.0
    
    def girar(self, vueltas):
        """
        Gira el objeto un número de vueltas.
        
        Args:
            vueltas (float): Número de vueltas a girar. Ej: 0.5 (180°), 1.5 (540°).
        
        Returns:
            float: Energía cinética generada (J).
                   Fórmula: E = 0.5 * I * ω²
                   Donde: I = momento de inercia (kg·m²), ω = velocidad angular (rad/s).
        
        Ejemplo:
            I = 1 kg·m², ω = 10 rad/s, vueltas = 0.5
            E = 0.5 * 1 * (10)² * 0.5 ≈ 25 J
        """
        self.angulo += 2 * np.pi * vueltas
        I = 1.0  # Momento de inercia (kg·m²)
        omega = (2 * np.pi * vueltas) / DT  # Velocidad angular (rad/s)
        self.energia_cinetica = 0.5 * I * omega**2
        return self.energia_cinetica


class DKP:
    """
    Clase para simular el DKP + Máquina Kilómetro.
    
    Atributos:
        objeto (Objeto): Objeto del DKP.
        lastre (Lastre): Lastre de la Máquina Kilómetro.
        E_perno_unitario (float): Energía por perno (J). Ej: 1.5 J.
        energia_total_generada (float): Energía total generada (J).
        energia_total_consumida (float): Energía total consumida (J).
    
    Métodos:
        subciclo_1(): Soltar lastre y tender hacia arriba.
        subciclo_2(vueltas): Giro y recorrer medio recorrido.
        subciclo_2_5(): Dejar subir del todo al objeto.
        subciclo_3(): Rotación de lastres por perneo.
        ciclo_completo(): Ejecuta los 4 subciclos.
    """
    
    def __init__(self, radio, altura_inicial, m_lastre, rho_lastre, g, eta_gen, eta_lift, E_perno_unitario):
        self.objeto = Objeto(radio, altura_inicial)
        self.lastre = Lastre(m_lastre, rho_lastre, g, altura_inicial)
        self.E_perno_unitario = E_perno_unitario
        self.energia_total_generada = 0.0
        self.energia_total_consumida = 0.0
    
    def subciclo_1(self):
        """
        Subciclo 1: Soltar lastre y tender hacia arriba (generar energía).
        
        Pasos:
            1. Soltar lastre desde Z = +45 a Z = -45.
            2. El objeto (sin lastre) tiende a subir por flotación, pero se mantiene a Z = 0.
            3. El objeto gira 180° (medio círculo).
        
        Returns:
            dict: Resultados del subciclo 1.
        """
        # Paso 1: Soltar lastre
        E_gen = self.lastre.soltar(ALTURA_FINAL)
        self.energia_total_generada += E_gen
        
        # Paso 2: Girar el objeto medio recorrido (180° = 0.5 vueltas)
        E_cin = self.objeto.girar(0.5)
        self.energia_total_generada += E_cin
        
        # Paso 3: El objeto tiende a subir, pero no del todo (se mantiene a Z = 0)
        self.objeto.altura = ALTURA_INICIAL - DELTA_H / 2
        
        return {
            'E_generada': E_gen + E_cin,
            'E_consumida': 0.0,
            'altura_objeto': self.objeto.altura,
            'altura_lastre': self.lastre.altura,
            'nota': 'Subciclo 1: Soltar lastre y girar 180°. Energía generada: E_gen + E_cin.'
        }
    
    def subciclo_2(self, vueltas=1.5):
        """
        Subciclo 2: Giro (invertir julios) y recorrer medio recorrido.
        
        Pasos:
            1. Girar el objeto 1.5 vueltas en sentido horario.
            2. El objeto se mantiene a Z = 0 (no sube del todo).
        
        Args:
            vueltas (float): Número de vueltas a girar (default: 1.5).
        
        Returns:
            dict: Resultados del subciclo 2.
        """
        # Paso 1: Girar el objeto 1.5 vueltas
        E_cin = self.objeto.girar(vueltas)
        self.energia_total_generada += E_cin
        
        # Paso 2: No dejar subir del todo (se mantiene a Z = 0)
        self.objeto.altura = ALTURA_INICIAL - DELTA_H / 2
        
        return {
            'E_generada': E_cin,
            'E_consumida': 0.0,
            'altura_objeto': self.objeto.altura,
            'altura_lastre': self.lastre.altura,
            'nota': 'Subciclo 2: Girar 1.5 vueltas. Energía generada: E_cin.'
        }
    
    def subciclo_2_5(self):
        """
        Subciclo 2.5: Dejar subir del todo al objeto.
        
        Pasos:
            1. Subir el objeto a Z = +45 (altura inicial).
        
        Returns:
            dict: Resultados del subciclo 2.5.
        """
        # Paso 1: Subir el objeto del todo (a Z = +45)
        self.objeto.altura = ALTURA_INICIAL
        
        return {
            'E_generada': 0.0,
            'E_consumida': 0.0,
            'altura_objeto': self.objeto.altura,
            'altura_lastre': self.lastre.altura,
            'nota': 'Subciclo 2.5: Subir objeto a Z = +45. No se genera ni consume energía.'
        }
    
    def subciclo_3(self):
        """
        Subciclo 3: Rotación de lastres por perneo.
        
        Pasos:
            1. Rotar lastres (el objeto recorre el recorrido en sentido antihorario).
            2. Perneo (4 pernos).
        
        Returns:
            dict: Resultados del subciclo 3.
        """
        # Paso 1: Rotación de lastres (energía cinética)
        E_cin = self.objeto.energia_cinetica
        self.energia_total_generada += E_cin
        
        # Paso 2: Perneo (4 pernos)
        E_perno = 4 * self.E_perno_unitario
        self.energia_total_consumida += E_perno
        
        # El objeto recorre el recorrido en sentido antihorario
        self.objeto.angulo -= 2 * np.pi * 0.5  # Medio recorrido en sentido opuesto
        
        return {
            'E_generada': E_cin,
            'E_consumida': E_perno,
            'altura_objeto': self.objeto.altura,
            'altura_lastre': self.lastre.altura,
            'nota': 'Subciclo 3: Rotar lastres y pernear. Energía generada: E_cin. Energía consumida: E_perno.'
        }
    
    def ciclo_completo(self):
        """
        Ejecuta los 4 subciclos completos.
        
        Returns:
            dict: Resultados del ciclo completo.
        """
        # Subciclo 1: Soltar lastre y tender hacia arriba
        resultado_c1 = self.subciclo_1()
        
        # Subciclo 2: Giro y recorrer medio recorrido
        resultado_c2 = self.subciclo_2()
        
        # Subciclo 2.5: Dejar subir del todo al objeto
        resultado_c2_5 = self.subciclo_2_5()
        
        # Subciclo 3: Rotación de lastres por perneo
        resultado_c3 = self.subciclo_3()
        
        # Energía total
        E_total_generada = self.energia_total_generada
        E_total_consumida = self.energia_total_consumida
        E_neta = E_total_generada - E_total_consumida
        
        # Eficiencia aparente (respecto a la energía potencial del lastre)
        E_entrada = self.lastre.m * self.lastre.g * DELTA_H
        eficiencia_aparente = (E_neta / E_entrada) * 100
        
        return {
            'E_generada': E_total_generada,
            'E_consumida': E_total_consumida,
            'E_neta': E_neta,
            'eficiencia_aparente': eficiencia_aparente,
            'subciclos': [resultado_c1, resultado_c2, resultado_c2_5, resultado_c3]
        }
