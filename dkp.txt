# --- Parámetros de energía ---
eta_gen = 0.85  # Eficiencia en generación
eta_lift = 0.90  # Eficiencia en elevación
E_perno = 1.5  # Energía por perno (J)

# --- Parámetros de simulación ---
dt = 0.01  # Paso de tiempo (s)
t_sim = 10.0  # Tiempo total de simulación (s)
t = np.arange(0, t_sim, dt)  # Vector de tiempo

# 🔬 CLASES PARA LOS CICLOS

class Lastre:
    """Clase para modelar el lastre en la Máquina Kilómetro."""
    
    def __init__(self, m, rho, g, altura_inicial):
        self.m = m  # Masa (kg)
        self.rho = rho  # Densidad (kg/m³)
        self.g = g  # Aceleración gravitatoria (m/s²)
        self.altura = altura_inicial  # Altura inicial (m)
        self.energia_potencial = self.m * self.g * self.altura  # Energía potencial inicial (J)
        
    def soltar(self, altura_final):
        """Suelta el lastre desde altura_inicial a altura_final."""
        delta_h = self.altura - altura_final
        energia_generada = self.m * self.g * delta_h * eta_gen
        self.altura = altura_final
        self.energia_potencial = self.m * self.g * self.altura
        return energia_generada
    
    def subir(self, altura_inicial):
        """Sube el lastre desde altura_final a altura_inicial."""
        delta_h = altura_inicial - self.altura
        energia_consumida = (self.m * self.g * delta_h) / eta_lift
        self.altura = altura_inicial
        self.energia_potencial = self.m * self.g * self.altura
        return energia_consumida


class Objeto:
    """Clase para modelar el objeto en el DKP."""
    
    def __init__(self, radio, altura_inicial, tiene_lastre=False):
        self.radio = radio  # Radio de giro (m)
        self.altura = altura_inicial  # Altura inicial (m)
        self.tiene_lastre = tiene_lastre  # ¿Tiene lastre?
        self.angulo = 0.0  # Ángulo inicial (rad)
        self.energia_cinetica = 0.0  # Energía cinética (J)
        
    def girar(self, vueltas):
        """Gira el objeto un número de vueltas."""
        self.angulo += 2 * np.pi * vueltas
        # Energía cinética (simplificación: I = m * r²)
        I = 1.0  # Momento de inercia (kg·m²)
        omega = (2 * np.pi * vueltas) / (dt * len(t))  # Velocidad angular (rad/s)
        self.energia_cinetica = 0.5 * I * omega**2
        return self.energia_cinetica
    
    def soltar_lastre(self):
        """Suelta el lastre del objeto."""
        self.tiene_lastre = False
        return self.energia_cinetica  # Devuelve la energía cinética almacenada


class DKP:
    """Clase para simular el DKP + Máquina Kilómetro."""
    
    def __init__(self, radio, altura_inicial, m_lastre, rho_lastre, g, eta_gen, eta_lift, E_perno):
        self.objeto = Objeto(radio, altura_inicial)
        self.lastre = Lastre(m_lastre, rho_lastre, g, altura_inicial)
        self.E_perno = E_perno
        self.energia_total_generada = 0.0
        self.energia_total_consumida = 0.0
        
    def ciclo_1(self):
        """
        Ciclo 1: Soltar lastre y tender hacia arriba (generar energía).
        - El lastre se suelta y el objeto tiende a subir por flotación.
        - Se recorre medio recorrido (180°).
        """
        # Soltar lastre
        E_gen = self.lastre.soltar(altura_final)
        self.energia_total_generada += E_gen
        
        # Girar el objeto medio recorrido (180° = 0.5 vueltas)
        E_cin = self.objeto.girar(0.5)
        self.energia_total_generada += E_cin
        
        # El objeto tiende a subir (pero no del todo)
        self.objeto.altura = altura_inicial - Delta_h / 2
        
        return {
            'E_generada': E_gen + E_cin,
            'E_consumida': 0.0,
            'altura_objeto': self.objeto.altura,
            'altura_lastre': self.lastre.altura
        }
    
    def ciclo_2(self, vueltas=1.5):
        """
        Ciclo 2: Giro (invertir julios) y hacer que el objeto recorra el recorrido que ha girado.
        - El objeto gira 1.5 vueltas en sentido horario.
        - No se deja subir del todo para mantener rendimiento positivo.
        """
        # Girar el objeto 1.5 vueltas
        E_cin = self.objeto.girar(vueltas)
        self.energia_total_generada += E_cin
        
        # El objeto no sube del todo (se mantiene a mitad de altura)
        self.objeto.altura = altura_inicial - Delta_h / 2
        
        return {
            'E_generada': E_cin,
            'E_consumida': 0.0,
            'altura_objeto': self.objeto.altura,
            'altura_lastre': self.lastre.altura
        }
    
    def ciclo_2_5(self):
        """
        Ciclo 2.5: Dejar subir del todo al objeto.
        - El objeto sube completamente a su altura inicial.
        """
        # Subir el objeto del todo
        self.objeto.altura = altura_inicial
        
        return {
            'E_generada': 0.0,
            'E_consumida': 0.0,
            'altura_objeto': self.objeto.altura,
            'altura_lastre': self.lastre.altura
        }
    
    def ciclo_3(self):
        """
        Ciclo 3: Rotación de lastres por perneo en el objeto.
        - Los lastres giran y hacen que el objeto recorra el recorrido en otro sentido.
        - Se despernean en el ciclo 1.
        """
        # Rotación de lastres (simplificación: se invierte la energía cinética)
        E_cin = self.objeto.energia_cinetica
        self.energia_total_generada += E_cin
        
        # Perneo (coste mínimo)
        E_perno_total = self.E_perno * 4  # 4 pernos
        self.energia_total_consumida += E_perno_total
        
        # El objeto recorre el recorrido en sentido antihorario
        self.objeto.angulo -= 2 * np.pi * 0.5  # Medio recorrido en sentido opuesto
        
        return {
            'E_generada': E_cin,
            'E_consumida': E_perno_total,
            'altura_objeto': self.objeto.altura,
            'altura_lastre': self.lastre.altura
        }
    
    def ciclo_completo(self):
        """Ejecuta los 4 ciclos completos."""
        # Ciclo 1: Soltar lastre y tender hacia arriba
        resultado_c1 = self.ciclo_1()
        
        # Ciclo 2: Giro y recorrer medio recorrido
        resultado_c2 = self.ciclo_2()
        
        # Ciclo 2.5: Dejar subir del todo al objeto
        resultado_c2_5 = self.ciclo_2_5()
        
        # Ciclo 3: Rotación de lastres y perneo
        resultado_c3 = self.ciclo_3()
        
        # Energía total
        E_total_generada = self.energia_total_generada
        E_total_consumida = self.energia_total_consumida
        E_neta = E_total_generada - E_total_consumida
        eficiencia_aparente = (E_neta / (self.lastre.m * self.lastre.g * Delta_h)) * 100
        
        return {
            'E_generada': E_total_generada,
            'E_consumida': E_total_consumida,
            'E_neta': E_neta,
            'eficiencia_aparente': eficiencia_aparente,
            'resultados': [resultado_c1, resultado_c2, resultado_c2_5, resultado_c3]
        }

    def __init__(self, m, rho, g, altura_inicial):
        self.m = m
        self.rho = rho
        self.g = g
        self.altura = altura_inicial
        self.energia_potencial = self.m * self.g * self.altura
    
    def soltar(self, altura_final):
        delta_h = self.altura - altura_final
        energia_generada = self.m * self.g * delta_h * eta_gen
        self.altura = altura_final
        self.energia_potencial = self.m * self.g * self.altura
        return energia_generada

class Objeto:
    def __init__(self, radio, altura_inicial, tiene_lastre=False):
        self.radio = radio
        self.altura = altura_inicial
        self.tiene_lastre = tiene_lastre
        self.angulo = 0.0
        self.energia_cinetica = 0.0
    
    def girar(self, vueltas):
        self.angulo += 2 * np.pi * vueltas
        I = 1.0
        omega = (2 * np.pi * vueltas) / 0.1  # dt = 0.1 s
        self.energia_cinetica = 0.5 * I * omega**2
        return self.energia_cinetica

class DKP:
    def __init__(self, radio, altura_inicial, m_lastre, rho_lastre, g, eta_gen, eta_lift, E_perno):
        self.objeto = Objeto(radio, altura_inicial)
        self.lastre = Lastre(m_lastre, rho_lastre, g, altura_inicial)
        self.E_perno = E_perno
        self.energia_total_generada = 0.0
        self.energia_total_consumida = 0.0
    
    def ciclo_1(self):
        E_gen = self.lastre.soltar(altura_final)
        self.energia_total_generada += E_gen
        E_cin = self.objeto.girar(0.5)
        self.energia_total_generada += E_cin
        self.objeto.altura = altura_inicial - Delta_h / 2
        return {'E_generada': E_gen + E_cin, 'E_consumida': 0.0}
    
    def ciclo_2(self, vueltas=1.5):
        E_cin = self.objeto.girar(vueltas)
        self.energia_total_generada += E_cin
        self.objeto.altura = altura_inicial - Delta_h / 2
        return {'E_generada': E_cin, 'E_consumida': 0.0}
    
    def ciclo_2_5(self):
        self.objeto.altura = altura_inicial
        return {'E_generada': 0.0, 'E_consumida': 0.0}
    
    def ciclo_3(self):
        E_cin = self.objeto.energia_cinetica
        self.energia_total_generada += E_cin
        E_perno_total = self.E_perno * 4
        self.energia_total_consumida += E_perno_total
        self.objeto.angulo -= 2 * np.pi * 0.5
        return {'E_generada': E_cin, 'E_consumida': E_perno_total}
    
    def ciclo_completo(self):
        self.ciclo_1()
        self.ciclo_2()
        self.ciclo_2_5()
        self.ciclo_3()
        E_neta = self.energia_total_generada - self.energia_total_consumida
        eficiencia = (E_neta / (self.lastre.m * self.lastre.g * Delta_h)) * 100
        return {
            'E_generada': self.energia_total_generada,
            'E_consumida': self.energia_total_consumida,
            'E_neta': E_neta,
            'eficiencia_aparente': eficiencia
        }
